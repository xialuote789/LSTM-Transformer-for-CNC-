class LSTMTransformer(nn.Module):
    """Full Teacher Model"""
    
    def __init__(self, input_dim=3, hidden_dim=128, lstm_layers=2,
                 transformer_layers=3, num_heads=8, dropout=0.2, num_classes=2):
        super().__init__()
        
        self.input_proj = nn.Linear(input_dim, hidden_dim)
        
        self.lstm = nn.LSTM(hidden_dim, hidden_dim, lstm_layers,
                          batch_first=True, dropout=dropout if lstm_layers > 1 else 0,
                          bidirectional=True)
        self.lstm_proj = nn.Linear(hidden_dim * 2, hidden_dim)
        
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim, nhead=num_heads,
            dim_feedforward=hidden_dim * 4, dropout=dropout,
            activation='gelu', batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, transformer_layers)
        
        self.classifier = nn.Sequential(
            nn.Linear(hidden_dim * 3, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_dim, num_classes)
        )
    
    def forward(self, x):
        x = self.input_proj(x)
        
        lstm_out, (h_n, _) = self.lstm(x)
        lstm_out_proj = self.lstm_proj(lstm_out)
        lstm_final = torch.cat([h_n[-2], h_n[-1]], dim=1)
        
        trans_out = self.transformer(lstm_out_proj)
        trans_final = trans_out.mean(dim=1)
        
        combined = torch.cat([lstm_final, trans_final], dim=1)
        output = self.classifier(combined)
        
        return output

class UltraLightLSTMTransformer(nn.Module):
    """Ultra-Light baseline model"""
    
    def __init__(self, input_dim=3, hidden_dim=64, lstm_layers=1,
                 transformer_layers=1, num_heads=4, dropout=0.15, num_classes=2):
        super().__init__()
        
        self.input_proj = nn.Linear(input_dim, hidden_dim)
        
        self.lstm = nn.LSTM(hidden_dim, hidden_dim, lstm_layers,
                          batch_first=True, bidirectional=True)
        
        self.lstm_proj = nn.Linear(hidden_dim * 2, hidden_dim)
        
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim, nhead=num_heads,
            dim_feedforward=hidden_dim * 2,
            dropout=dropout,
            activation='gelu', batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, transformer_layers)
        
        self.classifier = nn.Sequential(
            nn.Linear(hidden_dim * 3, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, num_classes)
        )
    
    def forward(self, x):
        x = self.input_proj(x)
        
        lstm_out, (h_n, _) = self.lstm(x)
        lstm_out_proj = self.lstm_proj(lstm_out)
        lstm_final = torch.cat([h_n[0], h_n[1]], dim=1)
        
        trans_out = self.transformer(lstm_out_proj)
        trans_final = trans_out.mean(dim=1)
        
        combined = torch.cat([lstm_final, trans_final], dim=1)
        output = self.classifier(combined)
        
        return output

class OnlyLSTM(nn.Module):
    """Ablation: Only LSTM, no Transformer"""
    
    def __init__(self, input_dim=3, hidden_dim=64, lstm_layers=1, 
                 dropout=0.15, num_classes=2):
        super().__init__()
        
        self.input_proj = nn.Linear(input_dim, hidden_dim)
        
        self.lstm = nn.LSTM(hidden_dim, hidden_dim, lstm_layers,
                          batch_first=True, bidirectional=True)
        
        self.classifier = nn.Sequential(
            nn.Linear(hidden_dim * 2, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_dim, num_classes)
        )
    
    def forward(self, x):
        x = self.input_proj(x)
        lstm_out, (h_n, _) = self.lstm(x)
        
        # Use final hidden states
        if h_n.shape[0] == 2:  # bidirectional
            lstm_final = torch.cat([h_n[0], h_n[1]], dim=1)
        else:
            lstm_final = h_n[-1]
        
        output = self.classifier(lstm_final)
        return output

class OnlyTransformer(nn.Module):
    """Ablation: Only Transformer, no LSTM"""
    
    def __init__(self, input_dim=3, hidden_dim=64, transformer_layers=1,
                 num_heads=4, dropout=0.15, num_classes=2):
        super().__init__()
        
        self.input_proj = nn.Linear(input_dim, hidden_dim)
        
        # Positional encoding
        self.pos_encoding = nn.Parameter(torch.randn(1, 2000, hidden_dim))
        
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim, nhead=num_heads,
            dim_feedforward=hidden_dim * 2,
            dropout=dropout,
            activation='gelu', batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, transformer_layers)
        
        self.classifier = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_dim // 2, num_classes)
        )
    
    def forward(self, x):
        x = self.input_proj(x)
        x = x + self.pos_encoding[:, :x.size(1), :]
        
        trans_out = self.transformer(x)
        trans_final = trans_out.mean(dim=1)  # Global average pooling
        
        output = self.classifier(trans_final)
        return output

class UnidirectionalLSTMTransformer(nn.Module):
    """Ablation: Unidirectional LSTM instead of Bidirectional"""
    
    def __init__(self, input_dim=3, hidden_dim=64, lstm_layers=1,
                 transformer_layers=1, num_heads=4, dropout=0.15, num_classes=2):
        super().__init__()
        
        self.input_proj = nn.Linear(input_dim, hidden_dim)
        
        # Unidirectional LSTM
        self.lstm = nn.LSTM(hidden_dim, hidden_dim, lstm_layers,
                          batch_first=True, bidirectional=False)  # Key change
        
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim, nhead=num_heads,
            dim_feedforward=hidden_dim * 2,
            dropout=dropout,
            activation='gelu', batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, transformer_layers)
        
        self.classifier = nn.Sequential(
            nn.Linear(hidden_dim * 2, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, num_classes)
        )
    
    def forward(self, x):
        x = self.input_proj(x)
        
        lstm_out, (h_n, _) = self.lstm(x)
        lstm_final = h_n[-1]
        
        trans_out = self.transformer(lstm_out)
        trans_final = trans_out.mean(dim=1)
        
        combined = torch.cat([lstm_final, trans_final], dim=1)
        output = self.classifier(combined)
        
        return output

class NoAttentionLSTM(nn.Module):
    """Ablation: LSTM with simple pooling instead of attention"""
    
    def __init__(self, input_dim=3, hidden_dim=64, lstm_layers=1,
                 dropout=0.15, num_classes=2):
        super().__init__()
        
        self.input_proj = nn.Linear(input_dim, hidden_dim)
        
        self.lstm = nn.LSTM(hidden_dim, hidden_dim, lstm_layers,
                          batch_first=True, bidirectional=True)
        
        # Simple pooling instead of attention
        self.pool = nn.AdaptiveAvgPool1d(1)
        
        self.classifier = nn.Sequential(
            nn.Linear(hidden_dim * 4, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_dim, num_classes)
        )
    
    def forward(self, x):
        x = self.input_proj(x)
        lstm_out, (h_n, _) = self.lstm(x)
        
        # Pooling over sequence
        pooled = self.pool(lstm_out.transpose(1, 2)).squeeze(-1)
        
        # Combine with hidden states
        lstm_final = torch.cat([h_n[0], h_n[1]], dim=1)
        combined = torch.cat([lstm_final, pooled], dim=1)
        
        output = self.classifier(combined)
        return output

class CNCDataset(Dataset):
    """CNC Dataset"""
    
    def __init__(self, X, y):
        self.X = torch.FloatTensor(X)
        self.y = torch.LongTensor(y)
    
    def __len__(self):
        return len(self.X)
    
    def __getitem__(self, idx):
        return self.X[idx], self.y[idx]

def create_realistic_synthetic_data():
    """Create realistic synthetic CNC vibration data for ablation testing"""
    print("Creating realistic synthetic data for ablation...")
    
    np.random.seed(42)
    
    def create_signal(n_samples, is_anomaly=False):
        t = np.linspace(0, 100, n_samples)
        
        # Normal vibration pattern
        signal = np.zeros((n_samples, 3))
        
        # X-axis
        signal[:, 0] = (
            0.5 * np.sin(2 * np.pi * 0.5 * t) +
            0.3 * np.sin(2 * np.pi * 2.0 * t) +
            0.1 * np.random.randn(n_samples)
        )
        
        # Y-axis
        signal[:, 1] = (
            0.3 * np.sin(2 * np.pi * 0.7 * t + np.pi/4) +
            0.2 * np.sin(2 * np.pi * 3.0 * t) +
            0.1 * np.random.randn(n_samples)
        )
        
        # Z-axis
        signal[:, 2] = (
            0.2 * np.sin(2 * np.pi * 1.0 * t + np.pi/2) +
            0.05 * np.random.randn(n_samples)
        )
        
        if is_anomaly:
            # Add anomaly patterns
            anomaly_start = np.random.randint(500, 1500)
            signal[anomaly_start:anomaly_start+200, :] *= np.random.uniform(2, 4)
            signal += np.random.randn(n_samples, 3) * 0.2
        
        return signal.astype(np.float32)
    
    # Create datasets
    X_train = []
    y_train = []
    X_val = []
    y_val = []
    
    # Training data
    for _ in range(1600):
        X_train.append(create_signal(2000, is_anomaly=False))
        y_train.append(0)
    for _ in range(400):
        X_train.append(create_signal(2000, is_anomaly=True))
        y_train.append(1)
    
    # Validation data
    for _ in range(400):
        X_val.append(create_signal(2000, is_anomaly=False))
        y_val.append(0)
    for _ in range(100):
        X_val.append(create_signal(2000, is_anomaly=True))
        y_val.append(1)
    
    X_train = np.array(X_train)
    y_train = np.array(y_train)
    X_val = np.array(X_val)
    y_val = np.array(y_val)
    
    # Normalize
    mean = X_train.mean(axis=(0, 1))
    std = X_train.std(axis=(0, 1))
    X_train = (X_train - mean) / (std + 1e-8)
    X_val = (X_val - mean) / (std + 1e-8)
    
    print(f"Created synthetic data: {len(X_train)} train, {len(X_val)} val")
    
    return X_train, X_val, y_train, y_val

def train_model(model, train_loader, val_loader, epochs=10, learning_rate=1e-3):
    """Train a model without distillation"""
    
    model = model.to(device)
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.AdamW(model.parameters(), lr=learning_rate)
    scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)
    scaler = GradScaler()
    
    best_val_acc = 0
    history = {'train_loss': [], 'train_acc': [], 'val_loss': [], 'val_acc': []}
    
    for epoch in range(1, epochs + 1):
        # Training
        model.train()
        train_loss = 0
        correct = 0
        total = 0
        
        for data, target in tqdm(train_loader, desc=f'Epoch {epoch}/{epochs}', leave=False):
            data, target = data.to(device), target.to(device)
            
            optimizer.zero_grad()
            
            with autocast(device_type='cuda', dtype=torch.float16):
                output = model(data)
                loss = criterion(output, target)
            
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            
            train_loss += loss.item()
            _, predicted = torch.max(output.data, 1)
            total += target.size(0)
            correct += (predicted == target).sum().item()
        
        train_loss /= len(train_loader)
        train_acc = correct / total
        
        # Validation
        model.eval()
        val_loss = 0
        correct = 0
        total = 0
        
        with torch.no_grad():
            for data, target in val_loader:
                data, target = data.to(device), target.to(device)
                
                with autocast(device_type='cuda', dtype=torch.float16):
                    output = model(data)
                    loss = criterion(output, target)
                
                val_loss += loss.item()
                _, predicted = torch.max(output.data, 1)
                total += target.size(0)
                correct += (predicted == target).sum().item()
        
        val_loss /= len(val_loader)
        val_acc = correct / total
        
        history['train_loss'].append(train_loss)
        history['train_acc'].append(train_acc)
        history['val_loss'].append(val_loss)
        history['val_acc'].append(val_acc)
        
        if val_acc > best_val_acc:
            best_val_acc = val_acc
        
        scheduler.step()
    
    return best_val_acc, history

def train_with_distillation(student_model, teacher_model, train_loader, val_loader, 
                           epochs=10, learning_rate=1e-3, temperature=4.0, alpha=0.7):
    """Train with knowledge distillation"""
    
    student_model = student_model.to(device)
    teacher_model = teacher_model.to(device)
    teacher_model.eval()
    
    ce_criterion = nn.CrossEntropyLoss()
    optimizer = optim.AdamW(student_model.parameters(), lr=learning_rate)
    scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)
    scaler = GradScaler()
    
    best_val_acc = 0
    history = {'train_loss': [], 'train_acc': [], 'val_loss': [], 'val_acc': []}
    
    for epoch in range(1, epochs + 1):
        # Training with distillation
        student_model.train()
        train_loss = 0
        correct = 0
        total = 0
        
        for data, target in tqdm(train_loader, desc=f'Distill Epoch {epoch}/{epochs}', leave=False):
            data, target = data.to(device), target.to(device)
            
            with torch.no_grad():
                teacher_outputs = teacher_model(data)
            
            optimizer.zero_grad()
            
            with autocast(device_type='cuda', dtype=torch.float16):
                student_outputs = student_model(data)
                
                # Distillation loss
                soft_targets = F.softmax(teacher_outputs / temperature, dim=1)
                soft_prob = F.log_softmax(student_outputs / temperature, dim=1)
                soft_loss = F.kl_div(soft_prob, soft_targets, reduction='batchmean') * (temperature ** 2)
                
                # Hard loss
                hard_loss = ce_criterion(student_outputs, target)
                
                # Combined loss
                loss = alpha * soft_loss + (1 - alpha) * hard_loss
            
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            
            train_loss += loss.item()
            _, predicted = torch.max(student_outputs.data, 1)
            total += target.size(0)
            correct += (predicted == target).sum().item()
        
        train_loss /= len(train_loader)
        train_acc = correct / total
        
        # Validation
        student_model.eval()
        val_loss = 0
        correct = 0
        total = 0
        
        with torch.no_grad():
            for data, target in val_loader:
                data, target = data.to(device), target.to(device)
                
                output = student_model(data)
                loss = ce_criterion(output, target)
                
                val_loss += loss.item()
                _, predicted = torch.max(output.data, 1)
                total += target.size(0)
                correct += (predicted == target).sum().item()
        
        val_loss /= len(val_loader)
        val_acc = correct / total
        
        history['train_loss'].append(train_loss)
        history['train_acc'].append(train_acc)
        history['val_loss'].append(val_loss)
        history['val_acc'].append(val_acc)
        
        if val_acc > best_val_acc:
            best_val_acc = val_acc
        
        scheduler.step()
    
    return best_val_acc, history

def run_architecture_ablation(train_loader, val_loader):
    """Test different architecture components"""
    
    print("\n" + "="*80)
    print("ARCHITECTURE ABLATION STUDY")
    print("="*80)
    
    models = {
        'Full Model (Ultra-Light)': UltraLightLSTMTransformer(),
        'Only LSTM': OnlyLSTM(),
        'Only Transformer': OnlyTransformer(),
        'Unidirectional LSTM': UnidirectionalLSTMTransformer(),
        'No Attention': NoAttentionLSTM()
    }
    
    results = {}
    
    for name, model in models.items():
        print(f"\nTraining: {name}")
        print("-" * 40)
        
        params = sum(p.numel() for p in model.parameters())
        print(f"Parameters: {params:,}")
        
        start_time = time.time()
        best_acc, history = train_model(model, train_loader, val_loader, epochs=10)
        train_time = time.time() - start_time
        
        results[name] = {
            'parameters': params,
            'best_val_acc': best_acc,
            'final_train_acc': history['train_acc'][-1],
            'training_time': train_time,
            'history': history
        }
        
        print(f"Best Val Acc: {best_acc:.4f}")
        print(f"Training Time: {train_time:.1f}s")
    
    return results

def main():
    """Main execution"""
    
    print("="*80)
    print("MODULE 6: COMPREHENSIVE ABLATION STUDY")
    print("="*80)
    
    # Load data
    print("\nStep 1: Loading data...")
    X_train, X_val, y_train, y_val = load_data()
    
    # Create datasets
    train_dataset = CNCDataset(X_train, y_train)
    val_dataset = CNCDataset(X_val, y_val)
    
    # Handle class imbalance if using real data
    if len(np.unique(y_train)) == 2:
        class_counts = np.bincount(y_train)
        if class_counts[0] / class_counts[1] > 2:  # Imbalanced
            class_weights = 1.0 / class_counts
            sample_weights = class_weights[y_train]
            sampler = WeightedRandomSampler(
                torch.DoubleTensor(sample_weights), 
                len(sample_weights)
            )
            train_loader = DataLoader(
                train_dataset, 
                batch_size=32, 
                sampler=sampler,
                num_workers=2,
                pin_memory=True
            )
        else:
            train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)
    else:
        train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)
    
    val_loader = DataLoader(val_dataset, batch_size=64, shuffle=False)
    
    # Run experiments
    print("\nStep 2: Running architecture ablation...")
    arch_results = run_architecture_ablation(train_loader, val_loader)
    
    print("\nStep 3: Running distillation ablation...")
    distill_results = run_distillation_ablation(train_loader, val_loader)
    
    # Print summary
    print("\n" + "="*80)
    print("ABLATION STUDY COMPLETE")
    print("="*80)
    
    print("\nArchitecture Ablation Results:")
    for name, result in arch_results.items():
        print(f"  {name}: {result['best_val_acc']:.4f}")
    
    print("\nDistillation Ablation Results:")
    for name, result in distill_results.items():
        print(f"  {name}: {result['best_val_acc']:.4f}")
    
    return arch_results, distill_results

if __name__ == "__main__":
    arch_results, distill_results = main()
