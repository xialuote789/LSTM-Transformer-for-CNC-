class CNCDataset(Dataset):
    """PyTorch数据集类"""

    def __init__(self, X, y):
        self.X = torch.FloatTensor(X)
        self.y = torch.LongTensor(y)

    def __len__(self):
        return len(self.X)

    def __getitem__(self, idx):
        return self.X[idx], self.y[idx]

def load_and_preprocess_data(data_path, max_files_per_class=10, window_size=2000, stride=1000):
    """加载和预处理数据"""
    print("\n" + "="*80)
    print("数据加载与预处理")
    print("="*80)

    data_path = Path(data_path)
    raw_data = []
    raw_labels = []

    # 加载数据
    for machine_dir in data_path.iterdir():
        if not machine_dir.is_dir():
            continue

        for process_dir in machine_dir.iterdir():
            if not process_dir.is_dir():
                continue

            # 加载good数据
            good_path = process_dir / "good"
            if good_path.exists():
                good_files = list(good_path.glob("*.h5"))[:max_files_per_class]
                for h5_file in tqdm(good_files,
                                  desc=f"加载 {machine_dir.name}/{process_dir.name}/good",
                                  leave=False):
                    try:
                        with h5py.File(h5_file, 'r') as f:
                            keys = list(f.keys())
                            data = f['vibration_data' if 'vibration_data' in keys else keys[0]][:]
                            raw_data.append(data)
                            raw_labels.append(0)
                    except:
                        pass

            # 加载bad数据
            bad_path = process_dir / "bad"
            if bad_path.exists():
                bad_files = list(bad_path.glob("*.h5"))[:max_files_per_class]
                for h5_file in tqdm(bad_files,
                                  desc=f"加载 {machine_dir.name}/{process_dir.name}/bad",
                                  leave=False):
                    try:
                        with h5py.File(h5_file, 'r') as f:
                            keys = list(f.keys())
                            data = f['vibration_data' if 'vibration_data' in keys else keys[0]][:]
                            raw_data.append(data)
                            raw_labels.append(1)
                    except:
                        pass

    print(f"✓ 加载了 {len(raw_data)} 个文件")

    # 创建滑动窗口
    windows = []
    labels = []

    for data, label in zip(raw_data, raw_labels):
        n_windows = (len(data) - window_size) // stride + 1
        if n_windows > 0:  # 确保有窗口可以创建
            for i in range(n_windows):
                start = i * stride
                end = start + window_size
                if end <= len(data):  # 确保不越界
                    windows.append(data[start:end])
                    labels.append(label)

    windows = np.array(windows, dtype=np.float32)
    labels = np.array(labels, dtype=np.int64)

    print(f"✓ 创建了 {len(windows)} 个窗口")
    print(f"  - 正常: {np.sum(labels == 0)}")
    print(f"  - 异常: {np.sum(labels == 1)}")

    # 分割数据集
    X_temp, X_test, y_temp, y_test = train_test_split(
        windows, labels, test_size=0.2, random_state=42, stratify=labels
    )
    X_train, X_val, y_train, y_val = train_test_split(
        X_temp, y_temp, test_size=0.1875, random_state=42, stratify=y_temp
    )

    # 标准化
    scaler = StandardScaler()
    X_train_flat = X_train.reshape(-1, 3)
    X_train_flat = scaler.fit_transform(X_train_flat)
    X_train = X_train_flat.reshape(X_train.shape)

    X_val_flat = X_val.reshape(-1, 3)
    X_val_flat = scaler.transform(X_val_flat)
    X_val = X_val_flat.reshape(X_val.shape)

    X_test_flat = X_test.reshape(-1, 3)
    X_test_flat = scaler.transform(X_test_flat)
    X_test = X_test_flat.reshape(X_test.shape)

    print(f"✓ 数据集分割:")
    print(f"  - 训练集: {len(X_train)} 样本")
    print(f"  - 验证集: {len(X_val)} 样本")
    print(f"  - 测试集: {len(X_test)} 样本")

    return X_train, X_val, X_test, y_train, y_val, y_test

def create_data_loaders(X_train, X_val, X_test, y_train, y_val, y_test, batch_size=32):
    """创建数据加载器"""
    # 创建数据集
    train_dataset = CNCDataset(X_train, y_train)
    val_dataset = CNCDataset(X_val, y_val)
    test_dataset = CNCDataset(X_test, y_test)

    # 处理类别不平衡
    class_counts = np.bincount(y_train)
    class_weights = 1.0 / class_counts
    sample_weights = class_weights[y_train]
    sample_weights = torch.DoubleTensor(sample_weights)

    train_sampler = WeightedRandomSampler(
        weights=sample_weights,
        num_samples=len(sample_weights),
        replacement=True
    )

    # 创建加载器
    train_loader = DataLoader(train_dataset, batch_size=batch_size,
                            sampler=train_sampler, num_workers=2, pin_memory=True)
    val_loader = DataLoader(val_dataset, batch_size=batch_size,
                          shuffle=False, num_workers=2, pin_memory=True)
    test_loader = DataLoader(test_dataset, batch_size=batch_size,
                           shuffle=False, num_workers=2, pin_memory=True)

    print(f"✓ 数据加载器创建完成 (batch_size={batch_size})")

    return train_loader, val_loader, test_loader

class PositionalEncoding(nn.Module):
    def __init__(self, d_model, dropout=0.1, max_len=5000):
        super().__init__()
        self.dropout = nn.Dropout(p=dropout)

        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(torch.arange(0, d_model, 2).float() *
                           (-math.log(10000.0) / d_model))

        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        pe = pe.unsqueeze(0).transpose(0, 1)
        self.register_buffer('pe', pe)

    def forward(self, x):
        x = x + self.pe[:x.size(0), :]
        return self.dropout(x)

class LSTMTransformerModel(nn.Module):
    def __init__(self, input_dim=3, hidden_dim=128, lstm_layers=2,
                 transformer_layers=3, num_heads=8, dropout=0.2, num_classes=2):
        super().__init__()

        # 输入投影
        self.input_proj = nn.Linear(input_dim, hidden_dim)

        # LSTM (双向)
        self.lstm = nn.LSTM(hidden_dim, hidden_dim, lstm_layers,
                          batch_first=True, dropout=dropout if lstm_layers > 1 else 0,
                          bidirectional=True)
        # LSTM输出投影: 2*hidden_dim -> hidden_dim
        self.lstm_proj = nn.Linear(hidden_dim * 2, hidden_dim)

        # Transformer
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim, nhead=num_heads,
            dim_feedforward=hidden_dim * 4, dropout=dropout,
            activation='gelu', batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, transformer_layers)

        # 分类器 - 修复：输入维度应该是 hidden_dim * 3 (lstm_final: hidden_dim*2, trans_final: hidden_dim)
        self.classifier = nn.Sequential(
            nn.Linear(hidden_dim * 3, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_dim, num_classes)
        )

    def forward(self, x):
        batch_size = x.size(0)

        # 输入投影: [batch, seq, 3] -> [batch, seq, hidden_dim]
        x = self.input_proj(x)

        # LSTM: 输入[batch, seq, hidden_dim], 输出[batch, seq, hidden_dim*2]
        lstm_out, (h_n, _) = self.lstm(x)
        # 投影回hidden_dim维度用于Transformer
        lstm_out_proj = self.lstm_proj(lstm_out)  # [batch, seq, hidden_dim]

        # 获取LSTM最终隐藏状态 (双向，所以要拼接最后两层)
        # h_n shape: [num_layers*2, batch, hidden_dim]
        lstm_final_forward = h_n[-2]  # 最后一层前向
        lstm_final_backward = h_n[-1]  # 最后一层后向
        lstm_final = torch.cat([lstm_final_forward, lstm_final_backward], dim=1)  # [batch, hidden_dim*2]

        # Transformer: 输入[batch, seq, hidden_dim]
        trans_out = self.transformer(lstm_out_proj)  # [batch, seq, hidden_dim]
        trans_final = trans_out.mean(dim=1)  # [batch, hidden_dim]

        # 合并特征: [batch, hidden_dim*2] + [batch, hidden_dim] = [batch, hidden_dim*3]
        combined = torch.cat([lstm_final, trans_final], dim=1)

        # 分类
        output = self.classifier(combined)

        return output

class Trainer:
    def __init__(self, model, train_loader, val_loader, test_loader,
                 learning_rate=1e-3, device=device):
        self.model = model.to(device)
        self.device = device
        self.train_loader = train_loader
        self.val_loader = val_loader
        self.test_loader = test_loader

        # 损失和优化器
        self.criterion = nn.CrossEntropyLoss()
        self.optimizer = optim.AdamW(model.parameters(), lr=learning_rate, weight_decay=1e-4)
        self.scheduler = CosineAnnealingWarmRestarts(self.optimizer, T_0=10, T_mult=2)
        self.scaler = GradScaler()

        # 记录
        self.history = {'train_loss': [], 'train_acc': [],
                       'val_loss': [], 'val_acc': []}
        self.best_val_acc = 0

    def train_epoch(self):
        self.model.train()
        total_loss = 0
        correct = 0
        total = 0

        pbar = tqdm(self.train_loader, desc='Training')
        for data, target in pbar:
            data, target = data.to(self.device), target.to(self.device)

            self.optimizer.zero_grad()

            # 混合精度训练（兼容版本）
            with autocast_mode():
                output = self.model(data)
                loss = self.criterion(output, target)

            self.scaler.scale(loss).backward()

            # 梯度裁剪
            self.scaler.unscale_(self.optimizer)
            torch.nn.utils.clip_grad_norm_(self.model.parameters(), max_norm=1.0)

            self.scaler.step(self.optimizer)
            self.scaler.update()

            total_loss += loss.item()
            _, predicted = torch.max(output.data, 1)
            total += target.size(0)
            correct += (predicted == target).sum().item()

            pbar.set_postfix({'loss': f'{loss.item():.4f}',
                            'acc': f'{correct/total:.4f}'})

        return total_loss / len(self.train_loader), correct / total

    def validate(self, loader, desc='Validating'):
        self.model.eval()
        total_loss = 0
        correct = 0
        total = 0
        all_preds = []
        all_labels = []

        with torch.no_grad():
            for data, target in tqdm(loader, desc=desc, leave=False):
                data, target = data.to(self.device), target.to(self.device)

                with autocast_mode():
                    output = self.model(data)
                    loss = self.criterion(output, target)

                total_loss += loss.item()
                _, predicted = torch.max(output.data, 1)
                total += target.size(0)
                correct += (predicted == target).sum().item()

                all_preds.extend(predicted.cpu().numpy())
                all_labels.extend(target.cpu().numpy())

        return total_loss / len(loader), correct / total, all_preds, all_labels

    def train(self, epochs=30):
        print("\n" + "="*80)
        print("开始训练")
        print("="*80)

        for epoch in range(1, epochs + 1):
            print(f"\nEpoch {epoch}/{epochs}")
            print("-" * 40)

            # 训练
            train_loss, train_acc = self.train_epoch()

            # 验证
            val_loss, val_acc, _, _ = self.validate(self.val_loader)

            # 记录
            self.history['train_loss'].append(train_loss)
            self.history['train_acc'].append(train_acc)
            self.history['val_loss'].append(val_loss)
            self.history['val_acc'].append(val_acc)

            # 学习率调整
            self.scheduler.step()
            current_lr = self.optimizer.param_groups[0]['lr']

            print(f"Train Loss: {train_loss:.4f}, Train Acc: {train_acc:.4f}")
            print(f"Val Loss: {val_loss:.4f}, Val Acc: {val_acc:.4f}")
            print(f"Learning Rate: {current_lr:.6f}")

            # 保存最佳模型
            if val_acc > self.best_val_acc:
                self.best_val_acc = val_acc
                torch.save({
                    'epoch': epoch,
                    'model_state_dict': self.model.state_dict(),
                    'optimizer_state_dict': self.optimizer.state_dict(),
                    'val_acc': val_acc,
                    'val_loss': val_loss
                }, 'best_model.pth')
                print(f"✓ 保存最佳模型 (Val Acc: {val_acc:.4f})")

            # GPU内存
            if torch.cuda.is_available():
                allocated = torch.cuda.memory_allocated() / 1024**2
                reserved = torch.cuda.memory_reserved() / 1024**2
                print(f"GPU内存: 已用 {allocated:.0f}MB / 预留 {reserved:.0f}MB")

        print("\n" + "="*80)
        print(f"训练完成！最佳验证准确率: {self.best_val_acc:.4f}")
        print("="*80)

    def test(self):
        print("\n" + "="*80)
        print("测试集评估")
        print("="*80)

        # 加载最佳模型
        checkpoint = torch.load('best_model.pth')
        self.model.load_state_dict(checkpoint['model_state_dict'])
        print(f"✓ 加载最佳模型 (Epoch {checkpoint['epoch']}, Val Acc: {checkpoint['val_acc']:.4f})")

        # 测试
        test_loss, test_acc, all_preds, all_labels = self.validate(
            self.test_loader, desc='Testing'
        )

        # 计算详细指标
        from sklearn.metrics import precision_recall_fscore_support
        precision, recall, f1, _ = precision_recall_fscore_support(
            all_labels, all_preds, average='binary', pos_label=1
        )

        print(f"\n测试集结果:")
        print(f"  准确率: {test_acc:.4f}")
        print(f"  精确率: {precision:.4f}")
        print(f"  召回率: {recall:.4f}")
        print(f"  F1分数: {f1:.4f}")

        # 分类报告
        print("\n分类报告:")
        print(classification_report(all_labels, all_preds,
                                  target_names=['Normal', 'Anomaly']))

        # 混淆矩阵
        self.plot_confusion_matrix(all_labels, all_preds)

        return test_acc, precision, recall, f1

    def plot_confusion_matrix(self, y_true, y_pred):
        cm = confusion_matrix(y_true, y_pred)

        plt.figure(figsize=(8, 6))
        sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
                   xticklabels=['Normal', 'Anomaly'],
                   yticklabels=['Normal', 'Anomaly'])
        plt.title('Confusion Matrix - Test Set')
        plt.ylabel('True Label')
        plt.xlabel('Predicted Label')

        # 添加更多信息
        total = cm.sum()
        accuracy = cm.diagonal().sum() / total
        plt.text(0.5, -0.12, f'Overall Accuracy: {accuracy:.4f}',
                ha='center', transform=plt.gca().transAxes, fontsize=12)

        plt.tight_layout()
        plt.show()

    def plot_history(self):
        fig, axes = plt.subplots(1, 2, figsize=(12, 4))

        # 损失曲线
        axes[0].plot(self.history['train_loss'], label='Train', linewidth=2)
        axes[0].plot(self.history['val_loss'], label='Val', linewidth=2)
        axes[0].set_xlabel('Epoch')
        axes[0].set_ylabel('Loss')
        axes[0].set_title('Loss Curve')
        axes[0].legend()
        axes[0].grid(True, alpha=0.3)

        # 准确率曲线
        axes[1].plot(self.history['train_acc'], label='Train', linewidth=2)
        axes[1].plot(self.history['val_acc'], label='Val', linewidth=2)
        axes[1].set_xlabel('Epoch')
        axes[1].set_ylabel('Accuracy')
        axes[1].set_title('Accuracy Curve')
        axes[1].legend()
        axes[1].grid(True, alpha=0.3)

        # 标记最佳点
        best_epoch = np.argmax(self.history['val_acc'])
        axes[1].plot(best_epoch, self.history['val_acc'][best_epoch],
                    'r*', markersize=15, label=f'Best: {self.history["val_acc"][best_epoch]:.4f}')
        axes[1].legend()

        plt.suptitle('Training History', fontsize=14)
        plt.tight_layout()
        plt.show()

def main():
    """完整训练流程"""

    # 参数设置
    DATA_PATH = "/content/drive/MyDrive/CNC_Machining/data"
    MAX_FILES_PER_CLASS = 5  # 可以增加到10或None处理所有文件
    BATCH_SIZE = 32
    EPOCHS = 20  # 可以增加到30-50
    LEARNING_RATE = 1e-3

    print("="*80)
    print("CNC刀具预测性维护 - 完整训练流程")
    print("="*80)

    # 1. 数据预处理
    print("\n步骤1: 数据加载与预处理")
    X_train, X_val, X_test, y_train, y_val, y_test = load_and_preprocess_data(
        DATA_PATH,
        max_files_per_class=MAX_FILES_PER_CLASS
    )

    # 2. 创建数据加载器
    print("\n步骤2: 创建数据加载器")
    train_loader, val_loader, test_loader = create_data_loaders(
        X_train, X_val, X_test, y_train, y_val, y_test,
        batch_size=BATCH_SIZE
    )

    # 3. 创建模型
    print("\n步骤3: 创建LSTM+Transformer模型")
    model = LSTMTransformerModel(
        input_dim=3,
        hidden_dim=128,
        lstm_layers=2,
        transformer_layers=3,
        num_heads=8,
        dropout=0.2
    )

    total_params = sum(p.numel() for p in model.parameters())
    trainable_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
    print(f"模型参数量: {total_params:,}")
    print(f"可训练参数: {trainable_params:,}")
    print(f"模型大小: {total_params * 4 / 1024 / 1024:.2f} MB")

    # 验证模型输出形状
    print("\n验证模型形状...")
    with torch.no_grad():
        dummy_input = torch.randn(2, 2000, 3).to(device)
        dummy_output = model.to(device)(dummy_input)
        print(f"✓ 输入形状: {dummy_input.shape}")
        print(f"✓ 输出形状: {dummy_output.shape}")

    # 4. 训练模型
    print("\n步骤4: 训练模型")
    trainer = Trainer(
        model=model,
        train_loader=train_loader,
        val_loader=val_loader,
        test_loader=test_loader,
        learning_rate=LEARNING_RATE
    )

    trainer.train(epochs=EPOCHS)

    # 5. 绘制训练历史
    print("\n步骤5: 绘制训练历史")
    trainer.plot_history()

    # 6. 测试评估
    print("\n步骤6: 测试集评估")
    accuracy, precision, recall, f1 = trainer.test()

    # 7. 总结
    print("\n" + "="*80)
    print("训练完成 - 最终结果")
    print("="*80)
    print(f"最佳验证准确率: {trainer.best_val_acc:.4f}")
    print(f"测试集准确率: {accuracy:.4f}")
    print(f"测试集F1分数: {f1:.4f}")

    # 判断是否需要轻量化
    if accuracy > 0.90:
        print("\n✨ 模型性能优秀！可以进行轻量化以提高部署效率")

    return trainer, model
